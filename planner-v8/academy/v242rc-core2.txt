    const out=clone(latest||{}), sections=['classes','students','attendance','memos','textbookLibrary','textbookRecords','tests','readingSessions','exams','tasks','notices','activityFeed'];

    Object.keys(local||{}).forEach(k=>{
      if(sections.includes(k)||['version','updatedAt','updatedBy'].includes(k))return;
