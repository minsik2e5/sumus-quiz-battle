 payload=legacyToServer(serverState);
      const d=await api('/data',{method:'PUT',body:JSON.stringify({data:payload,baseVersion:serverVersion,action:'autosave-v2'})});
      if(d.data){applyServerToLegacy(d.data);serverVersion=Number(d.data.version||serverVersion+1);}
      pendingLocal=false;lastSaveError='';setSync('ok');
    }catch(e){
      if(e.status===409&&retry){
        try{const latest=await api('/data');if(latest.data){serverState=clone(latest.data);serverVersion=Number(latest.data.version||serverVersion);await saveServerNow(false);return;}}catch{}
      }
      pendingLocal=false;lastSaveError=e.message;setSync('err','저장 오류');
      setTimeout(()=>{if(authToken)saveServerNow(false)},7000);
    }
  }
  function queueServerSave(){if(!authToken||applyingServer)return;pendingLocal=true;setSync('busy');clearTimeout(saveTimerV2);saveTimerV2=setTimeout(()=>saveServerNow(),SAVE_DELAY);}
